```python
import requests
import json
import time
import os
from typing import Optional, Dict, Any

'''
注意：非会员模式同步调用工作流，可能会因为工作流执行时间过长问题，导致无法获取到真正的数据。建议充值会员，使用会员模式
个人访问令牌：https://www.coze.cn/open/oauth/pats
'''


class CozeAPI:
    """Coze API客户端，用于调用Coze的工作流"""

    def __init__(self, api_key: str = None, workflow_id: str = None):
        self.api_key = api_key or 'pat_FdYI5dz17AqshOrvkTaZsB4nmK9ecQO0WlI9phbTU8qWUKLduWMv41CHyf7yh7yJ'
        self.workflow_id = workflow_id or '7675575485233463305'
        self.base_url = 'https://api.coze.cn'
        self.upload_url = f"{self.base_url}/v1/files/upload"
        self.run_url = f"{self.base_url}/v1/workflow/run"
        self.headers = {'Authorization': f'Bearer {self.api_key}', 'Content-Type': 'application/json'}

    def _get_save_path(self, task_id: str) -> str:
        """内部方法：生成视频保存路径"""
        result_dir = os.path.join('coze_results')
        os.makedirs(result_dir, exist_ok=True)
        filename = f"result_{task_id}_{int(time.time())}.mp4"
        return os.path.join(result_dir, filename)

    def download_video(self, image_url: str, task_id: str) -> Optional[str]:
        """下载视频到本地，返回跨平台路径"""
        try:
            save_path = self._get_save_path(task_id)
            resp = requests.get(image_url, stream=True, timeout=120)
            resp.raise_for_status()

            with open(save_path, 'wb') as f:
                for chunk in resp.iter_content(chunk_size=8192):
                    if chunk:
                        f.write(chunk)

            print(f"视频下载成功: {save_path}")
            return save_path.replace('\\', '/')
        except Exception as e:
            print(f"视频下载失败: {str(e)}")
            return None

    def upload_file(self, file_path: str) -> str:
        """上传文件到Coze并获取文件ID"""
        if not os.path.exists(file_path):
            raise FileNotFoundError(f"文件不存在: {file_path}")

        try:
            print(f"开始上传文件: {file_path}")
            with open(file_path, 'rb') as f:
                files = {'file': (os.path.basename(file_path), f)}
                resp = requests.post(
                    self.upload_url,
                    headers={'Authorization': f'Bearer {self.api_key}'},
                    files=files,
                    timeout=120
                )

            print(f"上传响应: {resp.status_code} | {resp.text}")
            result = resp.json()

            if resp.status_code == 200 and result.get('code') == 0 and 'data' in result:
                file_id = result['data'].get('id')
                if not file_id:
                    raise Exception(f"未提取到文件ID: {result}")
                print(f"文件上传成功，ID: {file_id}")
                return file_id
            raise Exception(f"文件上传失败: {result}")
        except Exception as e:
            print(f"文件上传异常: {str(e)}")
            raise

    def _extract_video_url(self, output_data: Any) -> Optional[str]:
        """从工作流输出中提取视频地址，兼容同步和异步返回结构"""
        try:
            if isinstance(output_data, str) and output_data.startswith(('http://', 'https://')):
                return output_data

            current = output_data
            for _ in range(3):
                if isinstance(current, str):
                    current = json.loads(current)
                    continue
                break

            if not isinstance(current, dict):
                return None

            output_value = current.get('output')
            if isinstance(output_value, str) and output_value.startswith(('http://', 'https://')):
                return output_value

            nested_output = current.get('Output')
            if nested_output is not None:
                nested_url = self._extract_video_url(nested_output)
                if nested_url:
                    return nested_url

            if isinstance(output_value, (dict, str)):
                return self._extract_video_url(output_value)
        except (json.JSONDecodeError, TypeError):
            return None

        return None

    def _parse_run_result(self, result: Dict[str, Any]) -> Dict[str, Any]:
        """解析工作流执行结果，兼容非会员同步和会员异步返回"""
        output_data = result.get('data')
        if output_data:
            video_url = self._extract_video_url(output_data)
            task_id = str(result.get('execute_id') or int(time.time() * 1000))

            if video_url:
                local_path = self.download_video(video_url, task_id)
                return {
                    'code': 0,
                    'msg': 'success',
                    'data': {
                        'status': 'success',
                        'task_id': task_id,
                        'workflow_id': self.workflow_id,
                        'result_video_url': video_url,
                        'local_video_path': local_path,
                        'raw_output': output_data
                    }
                }

            return {
                'code': 0,
                'msg': 'success',
                'data': {
                    'status': 'success',
                    'task_id': task_id,
                    'workflow_id': self.workflow_id,
                    'raw_output': output_data
                }
            }

        task_id = result.get('execute_id')
        if task_id:
            print(f"工作流已异步提交，任务ID: {task_id}")
            return {
                'code': 0,
                'msg': 'success',
                'data': {
                    'status': 'pending',
                    'task_id': task_id,
                    'workflow_id': self.workflow_id
                }
            }

        raise Exception(f"工作流返回中没有 data 或 execute_id: {result}")

    def run_workflow(self, source_image: str, target_image: str, is_async: bool = False) -> Dict[str, Any]:
        """运行Coze数字人工作流；非会员默认同步执行"""
        try:
            source_id, target_id = self.upload_file(source_image), self.upload_file(target_image)

            payload = {
                "workflow_id": self.workflow_id,
                "parameters": {
                    "voice": json.dumps({"file_id": source_id}),
                    "video": json.dumps({"file_id": target_id}),
                    "text": "执行数字人操作"
                },
                "app_id": "",
                "is_async": is_async
            }

            print(f"调用工作流 {self.workflow_id}，is_async={is_async}...")
            resp = requests.post(self.run_url, headers=self.headers, json=payload, timeout=300)

            if resp.status_code != 200:
                raise Exception(f"工作流调用失败: {resp.status_code} | {resp.text}")

            result = resp.json()
            if result.get('code') != 0:
                raise Exception(f"工作流调用失败: {result.get('msg')} | {result}")

            return self._parse_run_result(result)
        except Exception as e:
            print(f"工作流调用异常: {str(e)}")
            raise

    def get_workflow_result(self, task_id: str) -> Dict[str, Any]:
        """查询异步工作流执行结果"""
        try:
            print(f"查询任务 {task_id} 结果...")
            query_url = f"{self.base_url}/v1/workflows/{self.workflow_id}/run_histories/{task_id}"
            resp = requests.get(query_url, headers=self.headers, timeout=60)

            if resp.status_code != 200:
                raise Exception(f"查询失败: {resp.status_code} | {resp.text}")

            result = resp.json()
            if result.get('code') != 0:
                raise Exception(f"查询结果失败: {result.get('msg')} | {result}")

            data = result.get('data', [{}])[0]
            exec_status = data.get('execute_status', '')

            if exec_status == 'Success':
                return self._parse_success_result(data, task_id)
            if exec_status == 'Fail':
                return {'status': 'failed', 'error_message': '任务执行失败', 'task_id': task_id}
            return {'status': 'processing', 'task_id': task_id}
        except Exception as e:
            print(f"查询结果异常: {str(e)}")
            raise

    def _parse_success_result(self, data: Dict[str, Any], task_id: str) -> Dict[str, Any]:
        """解析异步任务成功结果"""
        try:
            video_url = self._extract_video_url(data.get('output', '{}'))

            if not video_url:
                return {'status': 'success', 'message': '任务成功但未返回视频', 'task_id': task_id}

            local_path = self.download_video(video_url, task_id)
            return {
                'status': 'success',
                'result_video_url': video_url,
                'local_video_path': local_path,
                'processing_time': data.get('processing_time', 0),
                'task_id': task_id
            }
        except (json.JSONDecodeError, KeyError) as e:
            print(f"解析输出失败: {str(e)}")
            return {'status': 'success', 'message': '任务成功但解析输出失败', 'task_id': task_id}


if __name__ == '__main__':
    try:
        coze_api = CozeAPI()
        voice_path, video_path = './lqz.mp3', './lqz.mp4'
        print("开始测试Coze数字人工作流（非会员同步模式）...")

        run_result = coze_api.run_workflow(voice_path, video_path, is_async=False)
        print(f"工作流执行结果: {run_result}")

        if run_result.get('code') == 0:
            data = run_result['data']
            status_map = {
                'success': f"测试成功！\nURL: {data.get('result_video_url')}\n本地路径: {data.get('local_video_path')}",
                'pending': f"任务已提交，任务ID: {data.get('task_id')}"
            }
            print(status_map.get(data.get('status'), f"未知状态: {data}"))
    except Exception as e:
        print(f"测试失败: {str(e)}", exc_info=True)
```

# 8 安装python3.13

```python
# 0 补充：
	默认centos的操作系统中，有 3.9.25的环境--》因为linux上很多软件是基于python写的--》必须有python解释器环境---》wget 下载文件的软件--》基于python写的
	我们的项目跑在python3.13上
    由此：等我们python3.13装完后：centos9机器上，有两个python解释器环境
    python---》python3.9.25---》pip---》是跟python3.9.25配对的
    python、pip   是 3.9.25
    
    python3.13---》python3.13--》pip3.13--》装第三方软件的
    python3.13、pip3.13 是我们自己装的
    

    
#1  源码安装python，依赖一些第三方zlib* libffi-devel
dnf install openssl-devel bzip2-devel expat-devel readline-devel sqlite-devel psmisc libffi-devel zlib* libffi-devel  -y

# 2 前往用户根目录
cd 

#2 下载  3.13.11 源码 服务器终端
# https://registry.npmmirror.com/binary.html?path=python/
wget https://registry.npmmirror.com/-/binary/python/3.13.11/Python-3.13.11.tgz # 阿里云提供的
# wget https://www.python.org/ftp/python/3.13.11/Python-3.13.11.tgz  # 官方的，国外，慢
    

#3  解压安装包
tar -xf Python-3.13.11.tgz

#4 进入目标文件
cd Python-3.13.11

#5  配置安装路径：/usr/local/python3
# 把3.13.11 编译安装到/usr/local/python311路径下
./configure --prefix=/usr/local/python313

#6  编译并安装,如果报错，说明缺依赖
# yum install openssl-devel bzip2-devel expat-devel gdbm-devel readline-devel sqlite-devel psmisc libffi-devel zlib* libffi-devel  -y
# make只是编译----》可执行文件，没有安装
# 类似于在win上下载了安装包，但是没安装
# make install 安装---》类似于在win上下了安装包，一路下一步安装了，指定安装位置---》/usr/local/python3
make &&  make install

# 7  建立软连接：/usr/local/python313路径不在环境变量，终端命令 python3，pip3
ln -s /usr/local/python313/bin/python3 /usr/bin/python3.13
ln -s /usr/local/python313/bin/pip3 /usr/bin/pip3.13

# 机器上有多个python和pip命令，对应关系如下
python       3.9      pip 
python3.13   3.13     pip3.13

#8  删除安装包与文件：[可以留着不删]
rm -rf /root/Python-3.13.1
rm -rf /root/Python-3.13.11.tgz
```
