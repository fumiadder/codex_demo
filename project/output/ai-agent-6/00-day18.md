# 9 配置admin访问

```python
# 0 能访问到服务器的admin后端了，但是没有样式

# 1 admin 没有样式
# 2 原因：
	uwsgi运行django项目，是不能访问到静态文件的--》之前本地测试环境是可以的
    下面报错：向http://47.116.189.221:8080/static/admin/simpleui-x/elementui/theme-chalk/index.css 访问不到
    index.css是静态文件，访问不到
    
# 3 解决--》使用nginx转发静态文件
```



```python
# 1  django中收集静态资源，使用nginx代理
# settings.py中加入   把静态资源收集到这个文件夹下

# STATIC_ROOT = BASE_DIR / "staticfiles"  # 这句话有了，不用动了 BASE_DIR：change_face_api

# 2 项目根目录下创建
cd /home/pro/change_face_api
mkdir staticfiles 

# 3 执行命令，收集静态问题---》把所有js，css图片，收集到staticfiles 文件夹下 
cd /home/pro/change_face_api
python3.11 manage.py collectstatic  # 收集完，文件夹下就有很多静态文件

# 4 上面我们nginx已经配置了代理静态文件

# 新增的配置静态文件
	vi /usr/local/nginx/conf/nginx.conf
	# 加入这两句---》把 nginx.conf 删除--》新建一个nginx.conf，粘贴进下面
      location /static {
            alias /home/pro/change_face_api/staticfiles;
       }
      location /media {
            alias /home/pro/change_face_api/media;
       }

 
 # 5 重启nginx
	nginx -s reload
```

**完整的nginx.conf配置**

```python
events {
    worker_connections  1024;
}
http {
    include       mime.types;
    default_type  application/octet-stream;
    sendfile        on;
	server {
        listen 8080;
        server_name  127.0.0.1;
        charset utf-8;
        location / {
            include uwsgi_params;
            uwsgi_pass 127.0.0.1:8000;
            uwsgi_param UWSGI_SCRIPT change_face_api.wsgi; 
            uwsgi_param UWSGI_CHDIR /home/pro/change_face_api/;
            }
       location /static {
            alias /home/pro/change_face_api/staticfiles;
       }
```

```python
# 9 修改nginx配置--》nginx.conf 文件-->变成了https访问了
events {
    worker_connections  1024;
}
http {
    include       mime.types;
    default_type  application/octet-stream;
    sendfile        on;
    client_max_body_size 20M;
	server {
    	listen 443 ssl;
     	ssl_certificate /usr/local/nginx/cert/www.liuqingzheng.top.pem;
     	ssl_certificate_key /usr/local/nginx/cert/www.liuqingzheng.top.key;
        server_name  liuqingzheng.top;
        location / {
            include uwsgi_params;
            uwsgi_pass 127.0.0.1:8000;
            uwsgi_param UWSGI_SCRIPT change_face_api.wsgi;
            uwsgi_param UWSGI_CHDIR /home/pro/change_face_api/;
            }

       location /static {
            alias /home/pro/change_face_api/staticfiles;
       }
      location /media {
            alias /home/pro/change_face_api/media;
       }
    }
}
```
